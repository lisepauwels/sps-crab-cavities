\section{Overview}
\begin{frame}{Overview}
\small
\begin{itemize}
  \item All loss maps in this deck: \textbf{26~GeV}, TCP/TCSM at \textbf{5$\sigma$}.
  \item Crab cavities: total voltage \textbf{1~MV}, frequency \textbf{400.787294~MHz}. The kicking
    plane (Horizontal / Vertical) is the top-level split below.
  \item Working point $Q_x = 20.13$, $Q_y = 20.18$; chromaticity $\xi_x = 0.405$, $\xi_y = 0.35$;
    normalised emittances $\epsilon_x = \epsilon_y = 2~\mu$m.
  \item Two loss-map types: \textbf{Betatron} (single-plane blow-up, excited Horizontally or
    Vertically) and \textbf{Off-momentum} (RF frequency sweep, DPpos or DPneg).
  \item Crab cavity phase scanned from $-179^\circ$ to $179^\circ$ in steps of $30^\circ$
    (13 values), with and without machine errors -- unlike the MD, no ``slow'' loss map was
    simulated.
\end{itemize}
\end{frame}
